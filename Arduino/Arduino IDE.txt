int ledPin = 13; // LED 接在第 13 腳

void setup() {
  Serial.begin(9600); // 開啟序列埠，通訊速率設為 9600
  pinMode(ledPin, OUTPUT);
  digitalWrite(ledPin, LOW); // 預設讓燈熄滅
}

void loop() {
  if (Serial.available() > 0) {
    char data = Serial.read(); // 讀取從 Unity 傳過來的資料
    
    if (data == '1') {
      digitalWrite(ledPin, HIGH); // 收到 '1' 就讓 LED 燈亮起
    }
  }
}
